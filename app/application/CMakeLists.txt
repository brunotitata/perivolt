cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(PeriVoltApplication LANGUAGES CXX)
  include(CTest)
endif()
find_package(Threads REQUIRED)
if(NOT TARGET perivolt_domain)
  add_library(perivolt_domain STATIC ../domain/battery.cpp)
  target_compile_features(perivolt_domain PUBLIC cxx_std_20)
endif()
add_library(perivolt_application STATIC monitor.cpp)
target_include_directories(perivolt_application PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(perivolt_application PUBLIC cxx_std_20)
target_link_libraries(perivolt_application PUBLIC perivolt_domain Threads::Threads)
if(APPLE)
  target_sources(perivolt_application PRIVATE
    ../adapters/attackshark-x11/platform/macos/producer_macos.cpp
    ../adapters/attackshark-x11/platform/macos/discovery_macos.cpp)
  target_link_libraries(perivolt_application PRIVATE "-framework IOKit" "-framework CoreFoundation")
endif()
if(MSVC)
  target_compile_options(perivolt_application PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(perivolt_application PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(application_integration_test tests/monitor_test.cpp)
  target_link_libraries(application_integration_test PRIVATE perivolt_application)
  if(MSVC)
    target_compile_options(application_integration_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(application_integration_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME application_integration_test COMMAND application_integration_test)
endif()
