cmake_minimum_required(VERSION 3.21)
project(PeriVoltPrototype VERSION 0.1.0 LANGUAGES CXX)

option(PERIVOLT_BUILD_UI "Build the Qt6 Widgets prototype (requires Qt6 SDK)" OFF)
include(CTest)

function(perivolt_compile_settings target)
  target_compile_features(${target} PUBLIC cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()

add_library(perivolt_domain STATIC domain/battery.cpp)
target_include_directories(perivolt_domain PUBLIC domain)
perivolt_compile_settings(perivolt_domain)

add_library(perivolt_x11_parser STATIC adapters/attackshark-x11/battery_parser.cpp)
target_include_directories(perivolt_x11_parser PUBLIC adapters/attackshark-x11)
perivolt_compile_settings(perivolt_x11_parser)

add_library(perivolt_presenter STATIC ui/presenter.cpp)
target_include_directories(perivolt_presenter PUBLIC ui)
target_link_libraries(perivolt_presenter PUBLIC perivolt_domain)
perivolt_compile_settings(perivolt_presenter)

# Backend owns this module; it remains buildable without Qt and native I/O tests.
add_subdirectory(application)

if(BUILD_TESTING)
  function(perivolt_test target source library)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE ${library})
    perivolt_compile_settings(${target})
    # Keep author assertions active even with Release/RelWithDebInfo.
    if(MSVC)
      target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
  endfunction()
  perivolt_test(domain_test domain/tests/battery_test.cpp perivolt_domain)
  perivolt_test(domain_lifecycle_test domain/tests/lifecycle_test.cpp perivolt_domain)
  perivolt_test(domain_source_time_test domain/tests/source_time_test.cpp perivolt_domain)
  perivolt_test(parser_test adapters/attackshark-x11/battery_parser_test.cpp perivolt_x11_parser)
  perivolt_test(presenter_test ui/tests/presenter_test.cpp perivolt_presenter)
  perivolt_test(macos_adapter_fake_test adapters/attackshark-x11/platform/macos/adapter_test.cpp perivolt_x11_parser)
  perivolt_test(macos_producer_fake_test adapters/attackshark-x11/platform/macos/producer_test.cpp perivolt_x11_parser)
endif()

if(PERIVOLT_BUILD_UI)

    endif()

  qt_add_executable(perivolt_monitor ui/main.cpp ui/monitor_window.cpp)
  target_link_libraries(perivolt_monitor PRIVATE Qt6::Widgets perivolt_presenter perivolt_application)
